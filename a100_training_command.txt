  --ipc=host \
  --security-opt seccomp=unconfined \

  -v /raid/seokmin/hf_cache:/hf_cache \
  -v /usr/local/cuda-12.8:/usr/local/cuda \

  -v /raid/seokmin/seokmin/robomme_policy_learning/assets:/app/assets \
